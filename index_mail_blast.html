Kode Google Apps Script
Deploy kode ini di script.google.com sebagai Web App
// ───────────────────────────────────────────── // MailBlast — Google Apps Script // // Deploy sebagai Web App: // Execute as: Me // Who has access: Anyone // // PENTING: Gunakan doGet() bukan doPost() // ───────────────────────────────────────────── var VALID_TOKEN = "GANTI_TOKEN_KAMU_DI_SINI"; function doGet(e) { try { var token = e.parameter.token; var action = e.parameter.action; var to = e.parameter.to; var subject = e.parameter.subject; var body = e.parameter.body; if (token !== VALID_TOKEN) { return respond("error", "Token tidak valid"); } if (action === "verify") { return respond("success", "Token valid"); } if (action === "send") { if (!to || !subject || !body) { return respond("error", "Data tidak lengkap"); } // Kirim sebagai HTML agar link bisa diklik GmailApp.sendEmail(to, subject, "", { htmlBody: body // body sudah HTML dari client (newline → <br> dilakukan di frontend) }); return respond("success", "Terkirim ke " + to); } return respond("error", "Action tidak dikenali"); } catch(err) { return respond("error", err.toString()); } } function respond(status, message) { return ContentService .createTextOutput(JSON.stringify({ status: status, message: message })) .setMimeType(ContentService.MimeType.JSON); }
